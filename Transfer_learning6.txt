# PRACTICAL 6: TRANSFER LEARNING USING VGG16
# Dataset: Caltech-101 | Framework: PyTorch
# Split: 50% Training, 25% Validation, 25% Testing

import os
import copy
import random
import numpy as np
import matplotlib.pyplot as plt
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader, Subset
from torchvision import datasets, transforms, models
from torchvision.models import VGG16_Weights
from sklearn.model_selection import train_test_split

SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)
if torch.cuda.is_available():
    torch.cuda.manual_seed_all(SEED)
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
PROJECT_DIR = os.path.dirname(os.path.abspath(__file__))
DATA_DIR = os.path.join(PROJECT_DIR, 'data')
OUTPUT_DIR = os.path.join(PROJECT_DIR, 'outputs')
os.makedirs(DATA_DIR, exist_ok=True)
os.makedirs(OUTPUT_DIR, exist_ok=True)
BATCH_SIZE = 8
NUM_WORKERS = 0
INITIAL_EPOCHS = 5
FINE_TUNE_EPOCHS = 3
PATIENCE = 2
print('=' * 65)
print('TRANSFER LEARNING USING VGG-16')
print('=' * 65)
print('PyTorch version:', torch.__version__)
print('Device:', device)
if device.type == 'cuda':
    print('GPU:', torch.cuda.get_device_name(0))
else:
    print('GPU not available. CPU will be used.')
image_transforms = {'train': transforms.Compose([transforms.Lambda(lambda img: img.convert('RGB')), transforms.RandomResizedCrop(size=256, scale=(0.8, 1.0)), transforms.RandomRotation(degrees=15), transforms.ColorJitter(), transforms.RandomHorizontalFlip(), transforms.CenterCrop(size=224), transforms.ToTensor(), transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225])]), 'eval': transforms.Compose([transforms.Lambda(lambda img: img.convert('RGB')), transforms.Resize(size=256), transforms.CenterCrop(size=224), transforms.ToTensor(), transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225])])}
print('\nImage transformations defined successfully!')
print('\nLoading Caltech-101 dataset...')
print('First download may take several minutes.')
train_base = datasets.Caltech101(root=DATA_DIR, target_type='category', transform=image_transforms['train'], download=True)
eval_base = datasets.Caltech101(root=DATA_DIR, target_type='category', transform=image_transforms['eval'], download=False)
total_size = len(train_base)
class_names = train_base.categories
n_classes = len(class_names)
targets = np.array(train_base.y)
print('\nDataset loaded successfully!')
print('Total images:', total_size)
print('Number of classes:', n_classes)
indices = np.arange(total_size)
train_indices, remaining_indices = train_test_split(indices, test_size=0.5, random_state=SEED, stratify=targets)
valid_indices, test_indices = train_test_split(remaining_indices, test_size=0.5, random_state=SEED, stratify=targets[remaining_indices])
train_data = Subset(train_base, train_indices)
valid_data = Subset(eval_base, valid_indices)
test_data = Subset(eval_base, test_indices)
print('\nDataset split completed!')
print('Training images:', len(train_data))
print('Validation images:', len(valid_data))
print('Testing images:', len(test_data))
dataloaders = {'train': DataLoader(train_data, batch_size=BATCH_SIZE, shuffle=True, num_workers=NUM_WORKERS, pin_memory=device.type == 'cuda'), 'valid': DataLoader(valid_data, batch_size=BATCH_SIZE, shuffle=False, num_workers=NUM_WORKERS, pin_memory=device.type == 'cuda'), 'test': DataLoader(test_data, batch_size=BATCH_SIZE, shuffle=False, num_workers=NUM_WORKERS, pin_memory=device.type == 'cuda')}
sample_images, sample_labels = next(iter(dataloaders['train']))
print('\nDataLoaders created successfully!')
print('Sample image shape:', sample_images.shape)
print('Expected image channels: 3 (RGB)')
print('\nLoading pretrained VGG16 model...')
model = models.vgg16(weights=VGG16_Weights.DEFAULT)
for param in model.parameters():
    param.requires_grad = False
n_inputs = model.classifier[6].in_features
model.classifier[6] = nn.Sequential(nn.Linear(n_inputs, 256), nn.ReLU(), nn.Dropout(0.4), nn.Linear(256, n_classes), nn.LogSoftmax(dim=1))
model = model.to(device)
print('\nVGG16 loaded successfully!')
print('Custom classifier added!')
print('\nClassifier architecture:')
print(model.classifier)
criterion = nn.NLLLoss()

def train_model(model, dataloaders, criterion, optimizer, num_epochs, stage_name, patience=2):
    best_weights = copy.deepcopy(model.state_dict())
    best_val_loss = float('inf')
    epochs_without_improvement = 0
    history = {'train_loss': [], 'valid_loss': [], 'train_acc': [], 'valid_acc': []}
    print('\n' + '=' * 65)
    print(stage_name)
    print('=' * 65)
    for epoch in range(num_epochs):
        print(f'\nEpoch {epoch + 1}/{num_epochs}')
        for phase in ['train', 'valid']:
            if phase == 'train':
                model.train()
            else:
                model.eval()
            running_loss = 0.0
            running_correct = 0
            total = 0
            for images, labels in dataloaders[phase]:
                images = images.to(device)
                labels = labels.to(device)
                optimizer.zero_grad(set_to_none=True)
                with torch.set_grad_enabled(phase == 'train'):
                    outputs = model(images)
                    loss = criterion(outputs, labels)
                    predictions = outputs.argmax(dim=1)
                    if phase == 'train':
                        loss.backward()
                        optimizer.step()
                batch_size = labels.size(0)
                running_loss += loss.item() * batch_size
                running_correct += (predictions == labels).sum().item()
                total += batch_size
            epoch_loss = running_loss / total
            epoch_acc = running_correct / total
            history[f'{phase}_loss'].append(epoch_loss)
            history[f'{phase}_acc'].append(epoch_acc)
            print(f'{phase.capitalize()} Loss: {epoch_loss:.4f} | Accuracy: {epoch_acc * 100:.2f}%')
            if phase == 'valid':
                if epoch_loss < best_val_loss:
                    best_val_loss = epoch_loss
                    best_weights = copy.deepcopy(model.state_dict())
                    epochs_without_improvement = 0
                else:
                    epochs_without_improvement += 1
        if epochs_without_improvement >= patience:
            print('\nEarly stopping triggered.')
            break
    model.load_state_dict(best_weights)
    print('\nBest validation loss:', round(best_val_loss, 4))
    return (model, history)
optimizer = optim.Adam(model.classifier[6].parameters(), lr=0.001)
model, history_stage1 = train_model(model=model, dataloaders=dataloaders, criterion=criterion, optimizer=optimizer, num_epochs=INITIAL_EPOCHS, stage_name='STAGE 1: TRAINING CUSTOM CLASSIFIER', patience=PATIENCE)
print('\nInitial classifier training completed!')
print('\nUnfreezing final convolutional block...')
for param in model.features[24:].parameters():
    param.requires_grad = True
for param in model.classifier[6].parameters():
    param.requires_grad = True
optimizer = optim.Adam(filter(lambda p: p.requires_grad, model.parameters()), lr=0.0001)
model, history_stage2 = train_model(model=model, dataloaders=dataloaders, criterion=criterion, optimizer=optimizer, num_epochs=FINE_TUNE_EPOCHS, stage_name='STAGE 2: FINE-TUNING VGG16', patience=PATIENCE)
print('\nFine-tuning completed!')
print('\n' + '=' * 65)
print('FINAL TEST EVALUATION')
print('=' * 65)
model.eval()
test_loss = 0.0
correct = 0
total = 0
with torch.no_grad():
    for images, labels in dataloaders['test']:
        images = images.to(device)
        labels = labels.to(device)
        outputs = model(images)
        loss = criterion(outputs, labels)
        predictions = outputs.argmax(dim=1)
        batch_size = labels.size(0)
        test_loss += loss.item() * batch_size
        correct += (predictions == labels).sum().item()
        total += batch_size
test_loss = test_loss / total
test_accuracy = correct / total
print('Test Loss:', round(test_loss, 4))
print('Test Accuracy:', round(test_accuracy * 100, 2), '%')
print('Correct predictions:', correct, '/', total)
combined_history = {}
for key in history_stage1:
    combined_history[key] = history_stage1[key] + history_stage2[key]
plt.figure(figsize=(12, 5))
plt.subplot(1, 2, 1)
plt.plot(combined_history['train_acc'], label='Training Accuracy')
plt.plot(combined_history['valid_acc'], label='Validation Accuracy')
plt.title('VGG16 Model Accuracy')
plt.xlabel('Epoch')
plt.ylabel('Accuracy')
plt.legend()
plt.grid(True)
plt.subplot(1, 2, 2)
plt.plot(combined_history['train_loss'], label='Training Loss')
plt.plot(combined_history['valid_loss'], label='Validation Loss')
plt.title('VGG16 Model Loss')
plt.xlabel('Epoch')
plt.ylabel('Loss')
plt.legend()
plt.grid(True)
plt.tight_layout()
plt.savefig(os.path.join(OUTPUT_DIR, 'performance_graphs.png'), dpi=300)
plt.show()
print('\nDisplaying sample predictions...')
images, labels = next(iter(dataloaders['test']))
model.eval()
with torch.no_grad():
    outputs = model(images.to(device))
    predictions = outputs.argmax(dim=1).cpu()
mean = torch.tensor([0.485, 0.456, 0.406]).view(3, 1, 1)
std = torch.tensor([0.229, 0.224, 0.225]).view(3, 1, 1)
plt.figure(figsize=(12, 8))
for i in range(min(6, len(images))):
    image = images[i] * std + mean
    image = image.permute(1, 2, 0).numpy()
    image = np.clip(image, 0, 1)
    predicted_label = class_names[predictions[i].item()]
    actual_label = class_names[labels[i].item()]
    plt.subplot(2, 3, i + 1)
    plt.imshow(image)
    plt.title(f'Predicted: {predicted_label}\nActual: {actual_label}')
    plt.axis('off')
plt.tight_layout()
plt.savefig(os.path.join(OUTPUT_DIR, 'sample_predictions.png'), dpi=300)
plt.show()
MODEL_PATH = os.path.join(OUTPUT_DIR, 'vgg16_caltech101.pth')
torch.save({'model_state_dict': model.state_dict(), 'class_names': class_names, 'test_accuracy': test_accuracy, 'test_loss': test_loss}, MODEL_PATH)
print('\nModel saved successfully:', MODEL_PATH)
print('\n' + '=' * 65)
print('ASSIGNMENT NO. 6 COMPLETED SUCCESSFULLY')
print('=' * 65)
print('Model: VGG16')
print('Framework: PyTorch')
print('Dataset: Caltech-101')
print('Split: 50% Training, 25% Validation, 25% Testing')
print('Transfer Learning: Completed')
print('Fine-tuning: Completed')
print('Test Accuracy:', round(test_accuracy * 100, 2), '%')
print('Performance graphs generated.')
print('Saved Model:', MODEL_PATH)
print('=' * 65)
